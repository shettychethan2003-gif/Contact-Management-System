
#include <iostream>
#include <fstream>
#include <vector>
#include <string>
#include <limits>
using namespace std;

// Contact Structure
struct Contact {
    string name;
    string phone;
    string email;
    string address;
};

// Contact Management Class
class ContactManager {
private:
    vector<Contact> contacts;

    // Save contacts to file
    void saveToFile() {
        ofstream file("contacts.txt");

        if (!file) {
            cout << "Error saving contacts!\n";
            return;
        }

        for (const auto &c : contacts) {
            file << c.name << "\n";
            file << c.phone << "\n";
            file << c.email << "\n";
            file << c.address << "\n";
        }

        file.close();
    }

    // Load contacts from file
    void loadFromFile() {
        ifstream file("contacts.txt");

        if (!file)
            return;

        Contact c;

        while (getline(file, c.name)) {
            if (!getline(file, c.phone) ||
                !getline(file, c.email) ||
                !getline(file, c.address)) {
                break;
            }

            contacts.push_back(c);
        }

        file.close();
    }

    // Find contact by phone number
    int findByPhone(string phone) {
        for (int i = 0; i < contacts.size(); i++) {
            if (contacts[i].phone == phone)
                return i;
        }

        return -1;
    }

public:

    // Constructor
    ContactManager() {
        loadFromFile();
    }

    // Add Contact
    void addContact() {
        Contact c;

        cout << "\nEnter Name: ";
        getline(cin, c.name);

        cout << "Enter Phone Number: ";
        getline(cin, c.phone);

        if (findByPhone(c.phone) != -1) {
            cout << "Phone number already exists!\n";
            return;
        }

        cout << "Enter Email: ";
        getline(cin, c.email);

        cout << "Enter Address: ";
        getline(cin, c.address);

        contacts.push_back(c);

        saveToFile();

        cout << "\nContact Added Successfully!\n";
    }

    // Display All Contacts
    void displayContacts() {
        if (contacts.empty()) {
            cout << "\nNo Contacts Available!\n";
            return;
        }

        cout << "\n========== CONTACT LIST ==========\n";

        for (int i = 0; i < contacts.size(); i++) {
            cout << "\nContact " << i + 1;
            cout << "\nName: " << contacts[i].name;
            cout << "\nPhone: " << contacts[i].phone;
            cout << "\nEmail: " << contacts[i].email;
            cout << "\nAddress: " << contacts[i].address;
            cout << "\n----------------------------------\n";
        }
    }

    // Search Contact by Name or Phone
    void searchContact() {
        string search;
        bool found = false;

        cout << "\nEnter Name or Phone to Search: ";
        getline(cin, search);

        for (const auto &c : contacts) {
            if (c.name.find(search) != string::npos ||
                c.phone.find(search) != string::npos) {

                cout << "\nContact Found!";
                cout << "\nName: " << c.name;
                cout << "\nPhone: " << c.phone;
                cout << "\nEmail: " << c.email;
                cout << "\nAddress: " << c.address;
                cout << "\n-------------------------\n";

                found = true;
            }
        }

        if (!found)
            cout << "\nContact Not Found!\n";
    }

    // Edit Contact
    void editContact() {
        string phone;

        cout << "\nEnter Phone Number of Contact to Edit: ";
        getline(cin, phone);

        int index = findByPhone(phone);

        if (index == -1) {
            cout << "\nContact Not Found!\n";
            return;
        }

        cout << "\nEnter New Name: ";
        getline(cin, contacts[index].name);

        cout << "Enter New Phone Number: ";
        string newPhone;
        getline(cin, newPhone);

        int existing = findByPhone(newPhone);

        if (existing != -1 && existing != index) {
            cout << "Phone number already belongs to another contact!\n";
            return;
        }

        contacts[index].phone = newPhone;

        cout << "Enter New Email: ";
        getline(cin, contacts[index].email);

        cout << "Enter New Address: ";
        getline(cin, contacts[index].address);

        saveToFile();

        cout << "\nContact Updated Successfully!\n";
    }

    // Delete Contact
    void deleteContact() {
        string phone;

        cout << "\nEnter Phone Number to Delete: ";
        getline(cin, phone);

        int index = findByPhone(phone);

        if (index == -1) {
            cout << "\nContact Not Found!\n";
            return;
        }

        contacts.erase(contacts.begin() + index);

        saveToFile();

        cout << "\nContact Deleted Successfully!\n";
    }

    // Main Menu
    void menu() {
        int choice;

        do {
            cout << "\n================================";
            cout << "\n     CONTACT MANAGEMENT SYSTEM";
            cout << "\n================================";
            cout << "\n1. Add Contact";
            cout << "\n2. Display All Contacts";
            cout << "\n3. Search Contact";
            cout << "\n4. Edit Contact";
            cout << "\n5. Delete Contact";
            cout << "\n6. Exit";
            cout << "\n================================";

            cout << "\nEnter Your Choice: ";

            if (!(cin >> choice)) {
                cin.clear();
                cin.ignore(
                    numeric_limits<streamsize>::max(), '\n'
                );
                cout << "\nInvalid Input!\n";
                continue;
            }

            cin.ignore(
                numeric_limits<streamsize>::max(), '\n'
            );

            switch (choice) {
                case 1:
                    addContact();
                    break;

                case 2:
                    displayContacts();
                    break;

                case 3:
                    searchContact();
                    break;

                case 4:
                    editContact();
                    break;

                case 5:
                    deleteContact();
                    break;

                case 6:
                    saveToFile();
                    cout << "\nContacts Saved Successfully!";
                    cout << "\nThank You for Using Contact Manager!\n";
                    break;

                default:
                    cout << "\nInvalid Choice! Try Again.\n";
            }

        } while (choice != 6);
    }
};

// Main Function
int main() {
    ContactManager manager;

    manager.menu();

    return 0;
}
